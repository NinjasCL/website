# Example .env — copy to .env and replace the placeholders with your credentials.
# The uploader uses SFTP over SSH, so FTP_PORT is the SSH port (usually 22),
# not the plain-FTP port 21.
FTP_USER=your-username
FTP_PASSWORD=your-password
FTP_HOST=ftp.your-host
FTP_PORT=22
FTP_ROUTE=/home/your-username/public_html
# If the SFTP account is chrooted to the home directory, use /public_html instead.
